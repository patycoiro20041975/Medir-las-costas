function getSplinePoint(pts, t) { const p = (pts.length - 1) * t; const intPoint = Math.floor(p); const weight = p - intPoint; const p0 = pts[Math.max(0, intPoint - 1)]; const p1 = pts[intPoint]; const p2 = pts[Math.min(pts.length - 1, intPoint + 1)]; const p3 = pts[Math.min(pts.length - 1, intPoint + 2)]; return { x: 0.5 * ((2 * p1.x) + (-p0.x + p2.x) * weight + (2 * p0.x - 5 * p1.x + 4 * p2.x - p3.x) * weight * weight + (-p0.x + 3 * p1.x - 3 * p2.x + p3.x) * weight * weight * weight), y: 0.5 * ((2 * p1.y) + (-p0.y + p2.y) * weight + (2 * p0.y - 5 * p1.y + 4 * p2.y - p3.y) * weight * weight + (-p0.y + 3 * p1.y - 3 * p2.y + p3.y) * weight * weight * weight) }; }
